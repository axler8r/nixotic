class Config:
    host:  str

def fetch(
    cfg: Config,
    path: str,
    retries: Optional[int] = None,
) -> bytes:
    """Fetch a resource with retry logic."""
    limit = retries if retries is not None else MAX_RETRIES
    url   = f"http://{cfg.host}:{cfg.port}{path}"
    for attempt in range(limit):
        try:
            return _do_get(url)
        except OSError as exc:
            if attempt == limit - 1:
                raise
            _log(f"retry {attempt}: {exc}")

# Solarized Light syntax example — Elixir

defmodule Cache do
  @moduledoc "Simple in-process key-value cache backed by ETS."
  @table :cache_table

  @spec start() :: {:ok, atom()}
  def start() do
    table = :ets.new(@table, [:set, :public])
    {:ok, table}
  end

  @spec get(term()) :: {:ok, term()} | :miss
  def get(key) do
    case :ets.lookup(@table, key) do
      [{_key, val}]  -> {:ok, val}
      []            -> :miss
    end
  end

  defp log(msg) do
    IO.puts("[cache] #{msg}")
  end
end

// Solarized Light syntax example — Rust

use std::collections::HashMap;
use std::fmt;

#[derive(Debug, Clone)]
pub struct Registry<T> {
    entries:  HashMap<String, T>,
    max_size: usize,
}

impl<T: Clone + fmt::Display> Registry<T> {
    pub fn new(max_size: usize) -> Self {
        Self { entries: HashMap::new(), max_size }
    }

    pub fn insert(&mut self, key: String, val: T) -> bool {
        if self.entries.len() >= self.max_size {
            return false;
        }
        self.entries.insert(key, val);
        true
    }

    pub fn get(&self, key: &str) -> Option<&T> {
        self.entries.get(key)
    }
}

macro_rules! log_entry {
    ($key:expr, $val:expr) => {
        println!("{}: {}", $key, $val)
    };
}

// Solarized Light syntax example — C#

using System;
using System.Collections.Generic;

namespace Nixotic.Cache;

/// <summary>Generic bounded registry.</summary>
[Serializable]
public sealed class Registry<T>
{
    private readonly Dictionary<string, T> _entries = new();
    private readonly int                   _maxSize;

    public Registry(int maxSize) => _maxSize = maxSize;

    public bool Insert(string key, T value)
    {
        if (_entries.Count >= _maxSize) return false;
        _entries[key] = value;
        return true;
    }

    public bool TryGet(string key, out T? value)
        => _entries.TryGetValue(key, out value);

    public int  Count   => _entries.Count;
    public bool IsFull => _entries.Count >= _maxSize;
}

# Solarized Light syntax example — zsh

export EDITOR="nvim"
export MAX_JOBS=8

fetch-and-build() {
  local repo=$(git rev-parse --show-toplevel)
  local branch=${1:-stable}

  if [[ -z "$repo" ]]; then
    eccho "error: not in a git repo" >&2
    return 1
  fi

  git fetch --prune
  git checkout "$branch" || return

  nix build --no-link --print-out-paths \
    | tee "$repo/build.log"
}

for host in ambul8r infer8r; do
  nixos-rebuild --flake ".#$host" switch
done

# Solarized Light syntax example — Nushell

const MAX_JOBS = 8

# Build a Nix flake for each host in the list
def build-hosts [
  hosts: list<string>
  --dry-run       # (-d) Skip actual build
  --jobs: int = MAX_JOBS
] -> table {
  hosts
    | each {|host|
        let flake = $".#($host)"
        if $dry_run {
          {host: $host, status: "dry-run"}
        } else {
          let result = nixos-rebuild --flake $flake switch
          {
            host:   $host,
            status: (if $result.exit_code == 0 { "ok" } else { "fail" }),
            jobs:   $jobs
          }
        }
      }
}

build-hosts ["ambul8r" "infer8r"] --jobs 4
